How is animation performance?

    
All animations use only  transform  and  opacity  properties, avoiding layout/paint triggers. Fully GPU-accelerated, smooth 60fps on mobile.

function getPreviewCSS(){
  var s=parseFloat(document.getElementById("sizeSlider").value);
  var sp=parseFloat(document.getElementById("speedSlider").value);
  var c=document.getElementById("customColor").value;
  return currentSpinner.css.replace(/40px|30px|200px/g,s+"px").replace(/12px/g,Math.round(s*0.3)+"px").replace(/10px/g,Math.round(s*0.25)+"px").replace(/8px/g,Math.round(s*0.2)+"px").replace(/6px/g,Math.round(s*0.15)+"px").replace(/4px/g,Math.round(s*0.1)+"px").replace(/3px/g,Math.round(s*0.08)+"px").replace(/2px/g,Math.round(s*0.05)+"px").replace(/#22d3ee/g,c).replace(/rgba\(34,211,238/g,"rgba("+hexToRgb(c));
}

function updatePreview(){
  var box=document.getElementById("previewContainer");
  var s=document.getElementById("sizeSlider").value;
  var sp=document.getElementById("speedSlider").value;
  var c=document.getElementById("customColor").value;
  currentColor=c;
  document.getElementById("sizeLabel").textContent=s+"px";
  document.getElementById("speedLabel").textContent=parseFloat(sp).toFixed(1)+"s";
  var style=document.getElementById("previewStyle");
  if(!style){style=document.createElement("style");style.id="previewStyle";document.head.appendChild(style)}
  style.textContent=getPreviewCSS().replace(/1s|1\.2s|1\.4s|0\.8s|1\.5s|2s/g,sp+"s");
  box.innerHTML=currentSpinner.html;
  var loaderEl=box.querySelector(".loader");
  if(loaderEl&¤tSpinner.anim){
    loaderEl.style.animation="none";
    loaderEl.offsetHeight;
    loaderEl.style.animation=currentSpinner.anim+" "+sp+"s linear infinite";
  }
  document.querySelectorAll(".color-preset").forEach(function(el){el.classList.remove("active")});
  document.querySelector('.color-preset[data-color="'+c+'"]')?.classList.add("active");
}

function updateCode(){
  var fmt=document.getElementById("codeFormat").value;
  var css=getPreviewCSS();
  if(fmt==="css"){
    document.getElementById("codeContent").textContent=css;
  }else{
    var html=currentSpinner.html;
    document.getElementById("codeContent").textContent="\n"+html+"\n\n/* CSS */\n"+css;
  }
}

function copyCode(){
  var code=document.getElementById("codeContent").textContent;
  navigator.clipboard.writeText(code).then(function(){showToast("Code copied to clipboard!")}).catch(function(){showToast("Copy failed, please select manually")});
}

  updatePreview();
}
